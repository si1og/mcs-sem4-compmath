x = [1.41, 2.31, 4.13, 5.31, 6.01];

function D = calcDivDiffTable(x, y)
    n = length(x);
    D = zeros(n, n+1);
    D(:, 1) = x';
    D(:, 2) = y';

    for j = 3:n+1
        for i = 1:n-j+2
            D(i, j) = (D(i+1, j-1) - D(i, j-1)) / (D(i+j-2, 1) - D(i, 1));
        end
    end
end

D = calcDivDiffTable(x, y);

disp('Divided Difference Table:');
disp(D);
